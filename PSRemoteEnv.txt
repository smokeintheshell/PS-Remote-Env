using System;
using System.Text;
using System.Collections.Generic;
using System.ComponentModel;
using System.Runtime.InteropServices;

public static class RemoteEnvironment
{
    [StructLayout(LayoutKind.Sequential)]
    struct PROCESS_BASIC_INFORMATION
    {
        public IntPtr Reserved1;
        public IntPtr PebBaseAddress;
        public IntPtr Reserved2_0;
        public IntPtr Reserved2_1;
        public IntPtr UniqueProcessId;
        public IntPtr Reserved3;
    }

    [DllImport("kernel32.dll", SetLastError = true)]
    static extern IntPtr OpenProcess(uint access, bool inherit, int pid);

    [DllImport("kernel32.dll", SetLastError = true)]
    static extern bool ReadProcessMemory(
        IntPtr process, IntPtr address,
        byte[] buffer, IntPtr size, out IntPtr bytesRead);

    [DllImport("kernel32.dll", SetLastError = true)]
    static extern bool IsWow64Process(IntPtr process, out bool wow64);

    [DllImport("kernel32.dll")]
    static extern bool CloseHandle(IntPtr handle);

    [DllImport("ntdll.dll")]
    static extern int NtQueryInformationProcess(
        IntPtr process, int infoClass,
        out PROCESS_BASIC_INFORMATION info,
        int size, out int returnLength);

    static byte[] ReadMemory(IntPtr handle, long address, int size)
    {
        byte[] buffer = new byte[size];
        IntPtr read;

        if (!ReadProcessMemory(handle, new IntPtr(address),
            buffer, new IntPtr(size), out read) ||
            read.ToInt64() != size)
            throw new Win32Exception(Marshal.GetLastWin32Error(),
                "ReadProcessMemory failed");

        return buffer;
    }

    static long ReadPointer(IntPtr handle, long address)
    {
        return BitConverter.ToInt64(ReadMemory(handle, address, 8), 0);
    }

    public static string[] Get(int pid)
    {
        if (IntPtr.Size != 8)
            throw new NotSupportedException("Requires 64-bit PowerShell.");

        // PROCESS_QUERY_INFORMATION | PROCESS_VM_READ
        IntPtr handle = OpenProcess(0x410, false, pid);

        if (handle == IntPtr.Zero)
            throw new Win32Exception(Marshal.GetLastWin32Error());

        try
        {
            bool wow64;
            if (!IsWow64Process(handle, out wow64))
                throw new Win32Exception(Marshal.GetLastWin32Error());

            if (wow64)
                throw new NotSupportedException("32-bit targets are not supported.");

            PROCESS_BASIC_INFORMATION info;
            int returnLength;

            int status = NtQueryInformationProcess(
                handle, 0, out info,
                Marshal.SizeOf(typeof(PROCESS_BASIC_INFORMATION)),
                out returnLength);

            if (status != 0)
                throw new Exception("NtQueryInformationProcess failed: 0x" +
                    status.ToString("X8"));

            // PEB.ProcessParameters
            long parameters = ReadPointer(handle,
                info.PebBaseAddress.ToInt64() + 0x20);

            // RTL_USER_PROCESS_PARAMETERS.Environment
            long environment = ReadPointer(handle, parameters + 0x80);

            if (environment == 0)
                return new string[0];

            List<byte> bytes = new List<byte>();
            int nullCount = 0;
            bool terminated = false;

            // Read UTF-16LE until double-null terminator.
            for (int offset = 0; offset < 1024 * 1024; offset += 2)
            {
                byte[] pair = ReadMemory(handle, environment + offset, 2);
                bytes.AddRange(pair);

                if (pair[0] == 0 && pair[1] == 0)
                    nullCount++;
                else
                    nullCount = 0;

                if (nullCount == 2)
                {
                    terminated = true;
                    break;
                }
            }

            if (!terminated)
                throw new Exception("Environment terminator not found.");

            string text = Encoding.Unicode.GetString(bytes.ToArray());

            return text.Split(new char[] { '\0' },
                StringSplitOptions.RemoveEmptyEntries);
        }
        finally
        {
            CloseHandle(handle);
        }
    }
}
